# 퀵정렬 (quicksort)

기준 원소를 하나 고르고, 그보다 작은 것과 큰 것으로 나눈 뒤 각각을 다시 정렬한다.
평균 O(n log n), 최악 O(n^2). 기준을 잘못 고르면 최악이 된다.

## 퀵정렬 파이썬

```python
def quicksort(a):
    if len(a) <= 1:
        return a
    기준 = a[len(a) // 2]
    작은 = [x for x in a if x < 기준]
    같은 = [x for x in a if x == 기준]
    큰 = [x for x in a if x > 기준]
    return quicksort(작은) + 같은 + quicksort(큰)
```

## 퀵정렬 자바스크립트

```javascript
function quicksort(a) {
  if (a.length <= 1) return a;
  const pivot = a[Math.floor(a.length / 2)];
  const lo = a.filter(x => x < pivot);
  const eq = a.filter(x => x === pivot);
  const hi = a.filter(x => x > pivot);
  return [...quicksort(lo), ...eq, ...quicksort(hi)];
}
```

## 퀵정렬 C

```c
void quicksort(int a[], int lo, int hi) {
    if (lo >= hi) return;
    int pivot = a[(lo + hi) / 2], i = lo, j = hi;
    while (i <= j) {
        while (a[i] < pivot) i++;
        while (a[j] > pivot) j--;
        if (i <= j) { int t = a[i]; a[i] = a[j]; a[j] = t; i++; j--; }
    }
    quicksort(a, lo, j);
    quicksort(a, i, hi);
}
```

## 퀵정렬 러스트

```rust
fn quicksort<T: PartialOrd + Clone>(a: &[T]) -> Vec<T> {
    if a.len() <= 1 { return a.to_vec(); }
    let pivot = a[a.len() / 2].clone();
    let lo: Vec<T> = a.iter().filter(|x| **x < pivot).cloned().collect();
    let eq: Vec<T> = a.iter().filter(|x| **x == pivot).cloned().collect();
    let hi: Vec<T> = a.iter().filter(|x| **x > pivot).cloned().collect();
    [quicksort(&lo), eq, quicksort(&hi)].concat()
}
```

# 이진 검색 (binary search)

정렬된 배열에서 가운데를 보고 반씩 줄여 나간다. O(log n).
배열이 정렬돼 있지 않으면 쓸 수 없다.

## 이진 검색 파이썬

```python
def binary_search(a, target):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == target:
            return mid
        if a[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
```

## 이진 검색 자바

```java
static int binarySearch(int[] a, int target) {
    int lo = 0, hi = a.length - 1;
    while (lo <= hi) {
        int mid = (lo + hi) >>> 1;
        if (a[mid] == target) return mid;
        if (a[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}
```

## 이진 검색 고

```go
func binarySearch(a []int, target int) int {
    lo, hi := 0, len(a)-1
    for lo <= hi {
        mid := (lo + hi) / 2
        switch {
        case a[mid] == target:
            return mid
        case a[mid] < target:
            lo = mid + 1
        default:
            hi = mid - 1
        }
    }
    return -1
}
```

# 너비 우선 탐색 (BFS)

시작점에서 가까운 것부터 큐로 훑는다. 최단 경로를 찾을 때 쓴다.

## 너비 우선 탐색 파이썬

```python
from collections import deque

def bfs(graph, start):
    seen, q, order = {start}, deque([start]), []
    while q:
        node = q.popleft()
        order.append(node)
        for nxt in graph.get(node, []):
            if nxt not in seen:
                seen.add(nxt)
                q.append(nxt)
    return order
```

## 너비 우선 탐색 자바스크립트

```javascript
function bfs(graph, start) {
  const seen = new Set([start]);
  const q = [start], order = [];
  while (q.length) {
    const node = q.shift();
    order.push(node);
    for (const nxt of graph[node] || []) {
      if (!seen.has(nxt)) { seen.add(nxt); q.push(nxt); }
    }
  }
  return order;
}
```
